        // Load Bookmarks from LocalStorage
        function getBookmarks() {
            const val = localStorage.getItem("code_bookmarks");
            return val ? JSON.parse(val) : [];
        }

        function saveBookmarks(arr) {
            localStorage.setItem("code_bookmarks", JSON.stringify(arr));
            updateBookmarkTabUI();
        }

        // Bookmark check / toggle
        function isBookmarked(id) {
            return getBookmarks().includes(id);
        }

        function toggleBookmark(id) {
            let bookmarks = getBookmarks();
            if (bookmarks.includes(id)) {
                bookmarks = bookmarks.filter(item => item !== id);
                showToast("Đã xóa khỏi Thư mục nhớ!", "danger");
            } else {
                bookmarks.push(id);
                const data = inspectorDb[id];
                const name = data ? data.title : id;
                showToast("Đã lưu " + name + " vào Thư mục nhớ!", "success");
            }
            saveBookmarks(bookmarks);
            updateBookmarkButtonUI(id);
        }

        function updateBookmarkButtonUI(id) {
            if (!id) {
                bookmarkActionBtn.classList.add("d-none");
                return;
            }
            bookmarkActionBtn.classList.remove("d-none");
            if (isBookmarked(id)) {
                bookmarkActionBtn.innerHTML = `<i class="fa-solid fa-star text-warning"></i> <span class="text-warning">Đã lưu (Click xóa)</span>`;
                bookmarkActionBtn.style.background = "rgba(255,193,7,0.15)";
            } else {
                bookmarkActionBtn.innerHTML = `<i class="fa-regular fa-star"></i> <span>Lưu thư mục nhớ</span>`;
                bookmarkActionBtn.style.background = "rgba(255,152,0,0.08)";
            }
        }

                toast.style.opacity = "0";

        // Scans all elements with data-inspect-id on the current page
        function scanPageElements() {
            tabContentComponents.innerHTML = "";
            const elements = document.querySelectorAll("[data-inspect-id]");
            const ids = new Set();
            
            elements.forEach(el => {
                const id = el.getAttribute("data-inspect-id");
                if (id) ids.add(id);
            });

            if (ids.size === 0) {
                tabContentComponents.innerHTML = `<div class="text-muted text-center p-3 small font-italic">Không tìm thấy cấu phần nào trên trang này.</div>`;
                return;
            }

            ids.forEach(id => {
                const data = inspectorDb[id];
                const name = data ? data.title : id;
                
                const item = document.createElement("button");
                item.className = "btn btn-sm btn-dark text-start text-white-50 border-0 py-1.5 px-3 rounded w-100 d-flex align-items-center justify-content-between transition-all";
                item.style.background = "rgba(255,255,255,0.03)";

                `;

                // Hovering menu element flashes outline on page
                item.addEventListener("mouseenter", () => {
                    const targetEl = document.querySelector(`[data-inspect-id="${id}"]`);
                    if (targetEl) {
                        targetEl.style.outline = "3px dashed #ff9800";
                        targetEl.style.outlineOffset = "3px";
                    }
                });
                item.addEventListener("mouseleave", () => {
                    const targetEl = document.querySelector(`[data-inspect-id="${id}"]`);
                    if (targetEl) {
                        targetEl.style.outline = "";
                        targetEl.style.outlineOffset = "";
                    }
                });

                // Click scrolls to component and inspects it
                item.addEventListener("click", () => {
                    inspectComponent(id);
                    const targetEl = document.querySelector(`[data-inspect-id="${id}"]`);
                    if (targetEl) {
                        targetEl.scrollIntoView({ behavior: 'smooth', block: 'center' });

                            if (flashCount > 5) {
                                clearInterval(interval);
                                targetEl.style.outline = "";
                            }
                        }, 180);
                    }
                });

                tabContentComponents.appendChild(item);

        function positionBadges() {
            if (!isActive) return;
            const badges = document.querySelectorAll(".inspector-element-badge");
            badges.forEach(badge => {
                const id = badge.getAttribute("data-target-id");
                const el = document.querySelector(`[data-inspect-id="${id}"]`);
                if (el) {
                    const rect = el.getBoundingClientRect();
                    if (rect.width === 0 || rect.height === 0) {
                        badge.style.display = "none";
                    } else {
                        badge.style.display = "flex";
                        // Position badge at the top-left of the target element with slight offset
                        const top = rect.top + window.scrollY - 10;
                        const left = rect.left + window.scrollX - 10;
                        badge.style.top = `${top}px`;
                        badge.style.left = `${left}px`;
                    }
                } else {
                    badge.style.display = "none";
                }
            });
        }

        function createInspectorBadges() {
            removeInspectorBadges();
            
            const elements = document.querySelectorAll("[data-inspect-id]");
            const createdIds = new Set();
            elements.forEach(el => {
                const id = el.getAttribute("data-inspect-id");
                if (!id || createdIds.has(id)) return;
                createdIds.add(id);
                
                const badge = document.createElement("div");
                badge.className = "inspector-element-badge";
                badge.setAttribute("data-target-id", id);
                badge.title = `Nhấp để xem code của: ${inspectorDb[id] ? inspectorDb[id].title : id}`;
                badge.innerHTML = `<i class="fa-solid fa-code"></i>`;
                
                // Click badge performs inspection
                badge.addEventListener("click", function (e) {
                    e.stopPropagation();
                    e.preventDefault();
                    inspectComponent(id);
                    el.scrollIntoView({ behavior: 'smooth', block: 'center' });
                    // Flash outline
                    let flashCount = 0;
                    const interval = setInterval(() => {
                        el.style.outline = flashCount % 2 === 0 ? "4px dashed #81c408" : "3px dashed #ff9800";
                        flashCount++;
                        if (flashCount > 5) {
                            clearInterval(interval);
                            el.style.outline = "";
                        }
                    }, 180);
                });
                
                // Badge hover highlights target element
                badge.addEventListener("mouseenter", function () {
                    el.style.outline = "3px dashed #ff9800";
                    el.style.outlineOffset = "3px";
                    el.style.boxShadow = "0 0 25px rgba(255, 152, 0, 0.4)";

        // Hover element scales up its badge to link them visually
        function handleElementHover(e) {
            e.stopPropagation();
            if (!isActive) return;

            const id = this.getAttribute("data-inspect-id");
            const badge = document.querySelector(`.inspector-element-badge[data-target-id="${id}"]`);
            if (badge) {
                badge.style.transform = "scale(1.25)";
            }
        }

        function handleElementHoverLeave(e) {
            e.stopPropagation();
            if (!isActive) return;

            const id = this.getAttribute("data-inspect-id");
            const badge = document.querySelector(`.inspector-element-badge[data-target-id="${id}"]`);
            if (badge) {
                badge.style.transform = "";
            }
        }

        // Intercept clicks on inspectable elements to view code (no auto-bookmark)
        function handleElementClick(e) {
            if (!isActive) return;
            
            e.preventDefault();
            e.stopPropagation();
            
            const id = this.getAttribute("data-inspect-id");
            if (id) {
                // Inspect component only
                inspectComponent(id);
            }
        }

        function bindInspectEvents() {
            const elements = document.querySelectorAll("[data-inspect-id]");
            elements.forEach(el => {
                el.addEventListener("mouseenter", handleElementHover);
                el.addEventListener("mouseleave", handleElementHoverLeave);
                el.addEventListener("click", handleElementClick, true); // Use capture phase to intercept before other click handlers
            });
        }

        function unbindInspectEvents() {
            const elements = document.querySelectorAll("[data-inspect-id]");
            elements.forEach(el => {
                el.removeEventListener("mouseenter", handleElementHover);
                el.removeEventListener("mouseleave", handleElementHoverLeave);
                el.removeEventListener("click", handleElementClick, true);
            });
        }

            if (!isActive || !lastSelectedId) return;

            const codeText = inspectCode.innerText;
            navigator.clipboard.writeText(codeText).then(function () {
                const originalText = copyBtn.innerHTML;
                copyBtn.innerHTML = `<i class="fa-solid fa-check text-success"></i> <span class="text-success">Copied!</span>`;
                setTimeout(function () {
                    copyBtn.innerHTML = originalText;
                }, 2000);
            }).catch(function (err) {
                console.error("Could not copy text: ", err);
            });
        });

        // Segment Tabs Event switching
        tabBtnComponents.addEventListener("click", () => {
            tabBtnComponents.classList.add("active-tab-style");
            tabBtnComponents.style.background = "rgba(255,255,255,0.08)";
            tabBtnBookmarks.classList.remove("active-tab-style");
            tabBtnBookmarks.style.background = "transparent";
            tabBtnBookmarks.style.color = "#6c757d";
            
            tabContentComponents.classList.remove("d-none");
            tabContentBookmarks.classList.add("d-none");
        });

        tabBtnBookmarks.addEventListener("click", () => {
            tabBtnBookmarks.classList.add("active-tab-style");
            tabBtnBookmarks.style.background = "rgba(255,255,255,0.08)";
            tabBtnComponents.classList.remove("active-tab-style");
            tabBtnComponents.style.background = "transparent";
            tabBtnComponents.style.color = "#6c757d";
            
            tabContentBookmarks.classList.remove("d-none");
            tabContentComponents.classList.add("d-none");
            updateBookmarkTabUI();
        });
        
        // Initial bookmarks check count
        bookmarkCountBadge.innerText = getBookmarks().length;
    });
